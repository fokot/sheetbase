import {
  createSheetbase, createIndexedDBStorage,
  createGoogleSheetsRemote, column
} from '@fokot/sheetbase';

const db = createSheetbase({
  accountId: googleUser.id,
  storage: createIndexedDBStorage('my-app'),
  remote: createGoogleSheetsRemote({
    getCredentials: () => ({
      accountId: googleUser.id,
      accessToken: tokenInMemory
    })
  })
});

const journal = db.table({
  spreadsheetId: sheetId,
  sheet: 'Journal',
  keyColumn: 'Date',
  columns: {
    walk: column.boolean('Walk'),
    pages: column.number('Pages')
  }
});

await journal.open({ create: true });
await journal.set('2026-10-27', { walk: true });

Text, číslo, boolean, JSON alebo vlastný formát. null vymaže bunku; 0 aj false sú hodnoty.

NEPREPOJENÉ S TRACKEROM

Osobné aplikácie

Dobrá východisková architektúra pre denníky, trackery a interné nástroje. Oprávnenia rieši Google účet a prístup k súboru.

Ručné zmeny v Sheete

Pozície riadkov sa pred zápisom načítajú znova. Premenovaný stĺpec alebo duplicitný kľúč zastaví synchronizáciu; súbežné zoradenie môže spôsobiť konflikt.

Súbežné zariadenia

Posledný úspešný zápis do bunky vyhrá. Knižnica neposkytuje distribuované transakcie ani riadkové bezpečnostné pravidlá.

Schéma a prihlásenie

Automatické migrácie schémy ani OAuth obrazovky tu zatiaľ nie sú. Súčasný tracker sa nemení a jeho dáta sa neimportujú.

↳

Google adaptér je overený simulovaným API, nie zápisom do tvojich tabuliek. Synchronizácia beží pri otvorenej aplikácii; cache samotnej stránky je úloha aplikácie. Pri viacerých kartách používa IndexedDB adaptér Web Locks.
